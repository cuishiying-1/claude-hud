/// 未收录的模型走透传（无 ≈ 标注），诚实降级。
pub fn builtin_pricing() -> PricingTable {
    let mut t = PricingTable::new();
    for (model, input, output) in [
        ("claude-opus-4-7", 5.0e-6, 25.0e-6),
        ("claude-opus-4-5", 5.0e-6, 25.0e-6),
        ("claude-sonnet-4-6", 3.0e-6, 15.0e-6),
        ("claude-sonnet-4-5", 3.0e-6, 15.0e-6),
        ("claude-haiku-4-5-20251001", 1.0e-6, 5.0e-6),
        ("claude-haiku-4-5", 1.0e-6, 5.0e-6),
        ("claude-3-5-sonnet", 3.0e-6, 15.0e-6),
        ("claude-3-5-haiku", 0.8e-6, 4.0e-6),
        ("claude-3-opus", 15.0e-6, 75.0e-6),
    ] {
        t.insert(
            model.into(),
            PriceEntry {
                input,
                output,
                cache_read: input * 0.1,
                cache_creation: input * 1.25,
            },
        );
    }
    t
}

/// 用户 [pricing] 覆盖内置表的合并视图（本批次查询唯一入口）。
pub fn merged_pricing(config: &AppConfig) -> PricingTable {
    let mut t = builtin_pricing();
    for (k, v) in &config.pricing {
        t.insert(k.clone(), v.clone());
    }
    t
}

- [ ] **Step 3: 实现 cache 权重**

`src/core/pricing.rs` realtime_cost 替换为：

```rust
/// ⑲+② 实时路径成本：stdin 累计 token（input/output/cache_read/cache_creation）
/// × 单价。cache 字段为 0/缺失 → 结果与旧公式一致（回归不变）；命中返回
/// (估算值, true)；未命中 [pricing]（含内置表）→ 透传官方 total_cost_usd；
/// 命中但 token 全 0 → 无数据可算 → 透传。
pub fn realtime_cost(data: &SessionData, pricing: &PricingTable) -> (f64, bool) {
    if let Some(price) = pricing.get(&data.model.id) {
        let t_in = data.context_window.total_input_tokens as f64;
        let t_out = data.context_window.total_output_tokens as f64;
        let t_cr = data.context_window.cache_read_input_tokens as f64;
        let t_cc = data.context_window.cache_creation_input_tokens as f64;
        if t_in > 0.0 || t_out > 0.0 || t_cr > 0.0 || t_cc > 0.0 {
            return (
                price.input * t_in
                    + price.output * t_out
                    + price.cache_read * t_cr
                    + price.cache_creation * t_cc,
                true,
            );
        }
    }
    (data.cost.total_cost_usd, false)
}
```

```rust
/// ④ 压缩临近检查（纯函数）：eta ≤ threshold 且冷却过期 → 触发并 mark_fired。
/// 复用 AlertCooldown（Compaction 键）跨进程去重；threshold 0 = 关闭。
/// eta None（数据不足/速率为 0）→ 不触发。
pub fn check_compaction(
    eta_minutes: Option<u64>,
    threshold_minutes: u64,
    cooldown_minutes: u64,
    cooldown: &mut AlertCooldown,
    now: u64,
) -> bool {
    if threshold_minutes == 0 {
        return false;
    }
    let Some(eta) = eta_minutes else { return false };
    if eta > threshold_minutes {
        return false;
    }
    let window = cooldown_minutes.saturating_mul(60);
    let last = cooldown.fired_at(AlertKind::Compaction);
    if last != 0 && now.saturating_sub(last) < window {
        return false;
    }
    cooldown.mark_fired(AlertKind::Compaction, now);
    true
}
```
